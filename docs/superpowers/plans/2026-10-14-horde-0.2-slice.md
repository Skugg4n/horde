# Horde 0.2: spelbar skiva, byggplan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Riksvägen dag 1 till 3 spelbar i webbläsaren: bygg vapen på positioner, lägg hinder på vägen, överlev vågorna, förlora och börja om dagen med behållen forskning.

**Architecture:** Spelreglerna (`src/sim/`) är rena funktioner utan DOM eller Three.js och körs i både webbläsare och Node (tester + balanssimulering). Grafiken (`src/render/`) läser simuleringens tillstånd och händelser varje bildruta och ritar; den ändrar aldrig reglerna. Gränssnittet (`src/ui/`) skickar kommandon till simuleringen. `src/main.js` är en tillståndsmaskin: meny → dagintro → bygg ↔ våg → resultat.

**Tech Stack:** Vanilla JS ES-moduler, Three.js 0.160.0 via importmap (cdn.jsdelivr.net), `node --test`, Python http.server för lokal körning. Inget byggsteg, inga npm-beroenden.

## Global Constraints

- Allt spelartext på svenska med å, ä, ö. Inga em-dash-tecken någonstans i text eller kod-kommentarer.
- Klarspråk i spelet: en text per sak, inga interna ord (inte "dps", "rof", "aggro"); siffror som "Skada 14", "Räckvidd 22 m".
- Version syns diskret i hörnet, källa `src/version.js`; höj vid varje commit som ändrar spelet. 0.2.0 när skivan är klar; delsteg 0.1.x.
- Simuleringen får aldrig importera från `src/render/`, `src/ui/` eller `three`.
- Sparfil: `localStorage['horde.save']`, JSON med `v`-fält. Radera aldrig sparfil vid fel; vid trasig fil: logga, behåll den under `horde.save.broken`, starta nytt.
- Grafik: bara modeller byggda i kod av byggsatsen i `src/render/kit.js`. Inga bildfiler. Samma sol, dis och gradering som `mockups/prov-1-3d.html`.
- Dagsljus, aldrig natt.
- Lokal körning: `python3 -m http.server 8141 --directory ~/Projects/horde` (launch-namn `horde`), öppna `http://localhost:8141/`.

## Koordinater och regler (gäller alla uppgifter)

- Värld: x längs vägen (horden kommer från +x), z tvärs, y upp. 1 enhet = 1 meter.
- Vägen är en polyline `map.path` ([[x,z],...]) från spawn till basen. Zombier har `d` = sträcka längs vägen (0 vid spawn) och `off` = sidoförskjutning (−3..3).
- Hinder ligger i **rutor** längs vägen: `map.cells` = lista av `d`-värden (rutans start), varje ruta 3 m lång, hela vägbredden.
- Simuleringen stegas med fast `dt = 1/30`. `step(sim, dt)` fyller `sim.events` (töms i början av varje steg).

```js
export function makeRng(seed) {
  let a = seed >>> 0;
  return function () {
    a = (a + 0x6D2B79F5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```
`src/sim/path.js`:
```js
export function pathLength(path) {
  let L = 0;
  for (let i = 1; i < path.length; i++) L += Math.hypot(path[i][0] - path[i - 1][0], path[i][1] - path[i - 1][1]);
  return L;
}
export function pointAt(path, d) {
  let rest = Math.max(0, d);
  for (let i = 1; i < path.length; i++) {
    const [ax, az] = path[i - 1], [bx, bz] = path[i];
    const seg = Math.hypot(bx - ax, bz - az);
    const dirX = (bx - ax) / seg, dirZ = (bz - az) / seg;
    if (rest <= seg || i === path.length - 1) {
      const k = Math.min(rest, seg);
      return { x: ax + dirX * k, z: az + dirZ * k, dirX, dirZ };
    }
    rest -= seg;
  }
}
```

```js
import { ZOMBIES } from '../data/zombies.js';
import { makeRng } from './rng.js';
import { pathLength, pointAt } from './path.js';

export const WAVE_INTEL = 2;      // forskning per klarad våg
export const CELL_LEN = 3;

export function createSim({ map, day, cash, research = [], seed = 1 }) {
  const dayCfg = map.days[day - 1];
  const res = new Set(research);
  return {
    map, day, dayCfg, phase: 'build', waveIndex: 0, cash, intelEarned: 0,
    baseHp: map.baseHp, zombies: [], events: [], time: 0, rng: makeRng(seed),
    research: res, nextId: 1, spawnQueue: [], len: pathLength(map.path),
    slots: map.slots.map(s => ({ id: s.id, x: s.x, z: s.z, locked: !!(s.locked && !res.has(s.locked)), weapon: null })),
    obstacles: {},
  };
}

export function zombiePos(sim, z) {
  const p = pointAt(sim.map.path, z.d);
  return { x: p.x - p.dirZ * z.off, z: p.z + p.dirX * z.off, dirX: p.dirX, dirZ: p.dirZ };
}

export function startWave(sim) {
  if (sim.phase !== 'build') return false;
  const wave = sim.dayCfg.waves[sim.waveIndex];
  sim.spawnQueue = [];
  for (const g of wave.groups) for (let i = 0; i < g.count; i++) sim.spawnQueue.push({ at: sim.time + g.delay + i * g.interval, type: g.type });
  sim.spawnQueue.sort((a, b) => a.at - b.at);
  sim.phase = 'wave';
  return true;
}

function spawn(sim, type) {
  const z = { id: sim.nextId++, type, hp: ZOMBIES[type].hp, maxHp: ZOMBIES[type].hp, d: 0, off: (sim.rng() - 0.5) * 6, wob: sim.rng() * 6, dead: false, deadT: 0, blockedBy: null };
  sim.zombies.push(z);
  sim.events.push({ t: 'spawn', id: z.id });
}

export function step(sim, dt) {
  sim.events = [];
  if (sim.phase !== 'wave') return;
  sim.time += dt;
  while (sim.spawnQueue.length && sim.spawnQueue[0].at <= sim.time) spawn(sim, sim.spawnQueue.shift().type);
  moveZombies(sim, dt);
  fireWeapons(sim, dt);   // Task 4
  for (const z of sim.zombies) if (z.dead) z.deadT += dt;
  sim.zombies = sim.zombies.filter(z => !(z.dead && z.deadT > 20));
  if (sim.baseHp <= 0) { sim.phase = 'lost'; sim.events.push({ t: 'lost' }); return; }
  if (!sim.spawnQueue.length && sim.zombies.every(z => z.dead)) {
    sim.intelEarned += WAVE_INTEL;
    sim.events.push({ t: 'waveCleared', index: sim.waveIndex, intel: WAVE_INTEL });
    sim.waveIndex++;
    if (sim.waveIndex >= sim.dayCfg.waves.length) { sim.phase = 'won'; sim.events.push({ t: 'won' }); }
    else sim.phase = 'build';
  }
}

function moveZombies(sim, dt) {
  for (const z of sim.zombies) {
    if (z.dead) continue;
    const def = ZOMBIES[z.type];
    const speed = def.speed * obstacleEffect(sim, z, dt); // Task 5; returnerar 1 tills dess
    z.d += speed * dt;
    z.wob += dt * speed;
    if (z.d >= sim.len) {
      sim.baseHp -= def.baseDamage;
      z.dead = true; z.breached = true; z.deadT = 99;
      sim.events.push({ t: 'breach', id: z.id, damage: def.baseDamage });
    }
  }
}

function fireWeapons() {}
function obstacleEffect() { return 1; }
```

export function placeObstacle(sim, cell, type) {
  if (sim.phase !== 'build') return { ok: false, reason: 'fel läge' };
  if (sim.obstacles[cell]) return { ok: false, reason: 'upptagen' };
  if (!obstacleUnlocked(type, sim.research)) return { ok: false, reason: 'ej upplåst' };
  const o = OBSTACLES[type];
  if (sim.cash < o.cost) return { ok: false, reason: 'pengar' };
  sim.cash -= o.cost;
  sim.obstacles[cell] = o.kind === 'slow'
    ? { type, durability: o.durability * (sim.research.has('wireTough') ? 2 : 1), maxDurability: o.durability * (sim.research.has('wireTough') ? 2 : 1) }
    : { type, hp: o.hp, maxHp: o.hp };
  return { ok: true };
}
export function removeObstacle(sim, cell) {
  const ob = sim.obstacles[cell];
  if (sim.phase !== 'build' || !ob) return { ok: false, refund: 0 };
  const whole = ob.hp === ob.maxHp && ob.durability === ob.maxDurability;
  const refund = whole ? Math.round(OBSTACLES[ob.type].cost * 0.5) : 0;
  sim.cash += refund; delete sim.obstacles[cell];
  return { ok: true, refund };
}
function obstacleEffect(sim, z, dt) {
  const def = ZOMBIES[z.type];
  let mul = 1;
  sim.map.cells.forEach((cd, i) => {
    const ob = sim.obstacles[i]; if (!ob) return;
    const o = OBSTACLES[ob.type];
    if (o.kind === 'slow' && z.d >= cd && z.d < cd + CELL_LEN) {
      mul = Math.min(mul, o.slow);
      ob.durability -= dt;
      if (ob.durability <= 0) { delete sim.obstacles[i]; sim.events.push({ t: 'obstacleBroken', cell: i, type: ob.type }); }
    }
    if (o.kind === 'block' && z.d >= cd - 0.6 && z.d <= cd) {
      mul = 0; z.d = Math.min(z.d, cd); z.attacking = true;
      ob.hp -= def.hitDamage * dt;
      sim.events.push({ t: 'obstacleHit', cell: i });
      if (ob.hp <= 0) { delete sim.obstacles[i]; sim.events.push({ t: 'obstacleBroken', cell: i, type: ob.type }); }
    }
  });
  if (mul > 0) z.attacking = false;
